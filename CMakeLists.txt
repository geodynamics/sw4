cmake_minimum_required(VERSION 3.18)
project(SW4 LANGUAGES C CXX Fortran)

set(SW4_BACKEND OPENMP CACHE STRING "Execution backend: OPENMP, CUDA, or HIP")
set_property(CACHE SW4_BACKEND PROPERTY STRINGS OPENMP CUDA HIP)
string(TOUPPER "${SW4_BACKEND}" SW4_BACKEND)
if(SW4_BACKEND STREQUAL CPU)
  set(SW4_BACKEND OPENMP)
endif()
if(NOT SW4_BACKEND MATCHES "^(OPENMP|CUDA|HIP)$")
  message(FATAL_ERROR "SW4_BACKEND must be OPENMP, CUDA, or HIP")
endif()
set(SW4_PRECISION double CACHE STRING "Floating-point precision: double or single")
set_property(CACHE SW4_PRECISION PROPERTY STRINGS double single)
if(NOT SW4_PRECISION MATCHES "^(double|single)$")
  message(FATAL_ERROR "SW4_PRECISION must be double or single")
endif()
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(USE_PROJ "Enable PROJ 6 or newer" OFF)
option(USE_HDF5 "Enable parallel HDF5" OFF)
option(USE_ZFP "Enable HDF5 ZFP compression" OFF)
option(USE_SZ "Enable HDF5 SZ compression" OFF)
option(USE_FFTW3 "Enable MPI FFTW random materials" OFF)
option(SW4_BUILD_MOPT "Build native OpenMP material inversion executable sw4mopt" OFF)
# Match the CUDA Make workflow. Host OpenMP affinity can constrain CUDA's
# progress threads, even with one OpenMP thread. HIP retains its Make default.
set(SW4_GPU_HOST_OPENMP_DEFAULT ON)
if(SW4_BACKEND STREQUAL CUDA)
  set(SW4_GPU_HOST_OPENMP_DEFAULT OFF)
endif()
option(SW4_GPU_HOST_OPENMP "Enable host OpenMP in GPU builds" ${SW4_GPU_HOST_OPENMP_DEFAULT})
option(SW4_CUDA_ARRAY_PREFETCH "Enable experimental per-array CUDA prefetch" OFF)
option(SW4_CUDA_POOL_PREFETCH "Prefetch Umpire pools before CUDA time stepping" OFF)
set(SW4_GPU_MPI_BUFFERS STAGED CACHE STRING "GPU MPI buffers: STAGED or MANAGED")
set_property(CACHE SW4_GPU_MPI_BUFFERS PROPERTY STRINGS STAGED MANAGED)
option(SW4_USE_UMPIRE "Use Umpire for GPU memory allocation" ON)
set(SW4_GPU_CXX_STANDARD 20 CACHE STRING "C++ standard required by installed RAJA/Umpire")
if((USE_ZFP OR USE_SZ) AND NOT USE_HDF5)
  message(FATAL_ERROR "USE_ZFP and USE_SZ require USE_HDF5=ON")
endif()

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${PROJECT_BINARY_DIR}/bin")
set(CMAKE_Fortran_MODULE_DIRECTORY "${PROJECT_BINARY_DIR}/include")
find_package(MPI REQUIRED COMPONENTS C CXX Fortran)
if(SW4_BACKEND STREQUAL OPENMP OR SW4_BUILD_MOPT OR SW4_GPU_HOST_OPENMP)
  find_package(OpenMP REQUIRED COMPONENTS CXX Fortran)
endif()
find_package(BLAS REQUIRED)
find_package(LAPACK REQUIRED)
include(FortranCInterface)
FortranCInterface_HEADER(FC.h MACRO_NAMESPACE "FC_")
include(cmake/SW4Sources.cmake)
file(GLOB SW4_QUADPACK_SOURCES CONFIGURE_DEPENDS src/quadpack/*.f)

add_library(sw4_dependencies INTERFACE)
target_link_libraries(sw4_dependencies INTERFACE MPI::MPI_CXX MPI::MPI_Fortran
  ${LAPACK_LIBRARIES} ${BLAS_LIBRARIES})
target_include_directories(sw4_dependencies INTERFACE "${PROJECT_SOURCE_DIR}/src"
  "${PROJECT_BINARY_DIR}")
if(SW4_PRECISION STREQUAL single)
  target_include_directories(sw4_dependencies INTERFACE "${PROJECT_SOURCE_DIR}/src/float")
else()
  target_include_directories(sw4_dependencies INTERFACE "${PROJECT_SOURCE_DIR}/src/double")
endif()
find_package(Git QUIET)
set(SW4_GIT_VERSION unknown)
if(GIT_FOUND)
  execute_process(COMMAND "${GIT_EXECUTABLE}" describe --abbrev=8 --dirty --always
    WORKING_DIRECTORY "${PROJECT_SOURCE_DIR}" OUTPUT_VARIABLE SW4_GIT_VERSION
    OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
endif()
string(TIMESTAMP SW4_BUILD_DATE "%d/%m/%Y")
cmake_host_system_information(RESULT SW4_BUILD_HOST QUERY HOSTNAME)
target_compile_definitions(sw4_dependencies INTERFACE CMAKE_FC_FUNC=1
  EW_MADEBY="$ENV{USER}" EW_OPT_LEVEL="${CMAKE_BUILD_TYPE}"
  EW_COMPILER="${CMAKE_CXX_COMPILER}" EW_LIBDIR="NA" EW_INCDIR="NA"
  EW_HOSTNAME="${SW4_BUILD_HOST}" EW_WHEN="${SW4_BUILD_DATE}" SW4_GIT_VERSION="${SW4_GIT_VERSION}")

if(USE_PROJ)
  find_package(PROJ REQUIRED CONFIG)
  if(PROJ_VERSION VERSION_LESS 6.0)
    message(FATAL_ERROR "SW4 requires PROJ 6 or newer")
  endif()
  target_link_libraries(sw4_dependencies INTERFACE PROJ::proj)
  target_compile_definitions(sw4_dependencies INTERFACE ENABLE_PROJ ENABLE_PROJ_6)
endif()
if(USE_HDF5)
  set(HDF5_PREFER_PARALLEL TRUE)
  find_package(HDF5 REQUIRED COMPONENTS C HL)
  if(NOT HDF5_IS_PARALLEL)
    message(FATAL_ERROR "SW4 requires parallel HDF5; select a parallel HDF5 installation")
  endif()
  target_include_directories(sw4_dependencies INTERFACE ${HDF5_INCLUDE_DIRS})
  target_link_libraries(sw4_dependencies INTERFACE ${HDF5_LIBRARIES} ${HDF5_HL_LIBRARIES})
  target_compile_definitions(sw4_dependencies INTERFACE USE_HDF5)
endif()
if(USE_ZFP)
  set(H5Z_ZFP_USE_STATIC_LIBS ON)
  find_package(zfp REQUIRED CONFIG)
  find_package(H5Z_ZFP 1.1.0 REQUIRED CONFIG)
  target_link_libraries(sw4_dependencies INTERFACE h5z_zfp::h5z_zfp)
  target_compile_definitions(sw4_dependencies INTERFACE USE_ZFP)
endif()
if(USE_SZ)
  find_path(SZ_INCLUDE_DIR sz.h HINTS "${SZ_ROOT}/include")
  find_library(SZ_LIBRARY NAMES SZ sz HINTS "${SZ_ROOT}/lib" "${SZ_ROOT}/lib64")
  if(NOT SZ_INCLUDE_DIR OR NOT SZ_LIBRARY)
    message(FATAL_ERROR "USE_SZ requires SZ; set SZ_ROOT or SZ_INCLUDE_DIR/SZ_LIBRARY")
  endif()
  target_include_directories(sw4_dependencies INTERFACE "${SZ_INCLUDE_DIR}")
  target_link_libraries(sw4_dependencies INTERFACE "${SZ_LIBRARY}")
  target_compile_definitions(sw4_dependencies INTERFACE USE_SZ)
endif()
if(USE_FFTW3)
  find_path(FFTW3_INCLUDE_DIR fftw3-mpi.h HINTS "${FFTW_ROOT}/include" "$ENV{FFTW_ROOT}/include")
  if(SW4_PRECISION STREQUAL single)
    set(SW4_FFTW_NAME fftw3f)
  else()
    set(SW4_FFTW_NAME fftw3)
  endif()
  find_library(FFTW3_LIBRARY NAMES ${SW4_FFTW_NAME} HINTS "${FFTW_ROOT}/lib" "$ENV{FFTW_ROOT}/lib")
  find_library(FFTW3_MPI_LIBRARY NAMES ${SW4_FFTW_NAME}_mpi HINTS "${FFTW_ROOT}/lib" "$ENV{FFTW_ROOT}/lib")
  if(NOT FFTW3_INCLUDE_DIR OR NOT FFTW3_LIBRARY OR NOT FFTW3_MPI_LIBRARY)
    message(FATAL_ERROR "USE_FFTW3 requires precision-matched MPI FFTW libraries; set FFTW_ROOT")
  endif()
  target_include_directories(sw4_dependencies INTERFACE "${FFTW3_INCLUDE_DIR}")
  target_link_libraries(sw4_dependencies INTERFACE "${FFTW3_MPI_LIBRARY}" "${FFTW3_LIBRARY}")
  target_compile_definitions(sw4_dependencies INTERFACE USE_FFTW ENABLE_FFTW)
endif()

function(sw4_native_library target)
  add_library(${target} STATIC ${SW4_NATIVE_CXX_SOURCES} ${SW4_NATIVE_FORTRAN_SOURCES} ${SW4_QUADPACK_SOURCES})
  target_link_libraries(${target} PUBLIC sw4_dependencies OpenMP::OpenMP_CXX OpenMP::OpenMP_Fortran)
  target_compile_features(${target} PUBLIC cxx_std_11)
endfunction()

if(SW4_BACKEND STREQUAL OPENMP OR SW4_BUILD_MOPT)
  sw4_native_library(sw4_native)
endif()
if(SW4_BACKEND STREQUAL OPENMP)
  add_executable(sw4 src/main.C)
  target_link_libraries(sw4 PRIVATE sw4_native)
else()
  include(cmake/SW4GPU.cmake)
endif()
if(SW4_BUILD_MOPT)
  add_executable(sw4mopt src/moptmain.C src/solve-backward-allpars.C src/lbfgs.C
    src/nlcg.C src/ProjectMtrl.C src/MaterialParameterization.C src/Mopt.C
    src/MaterialParCartesian.C src/InterpolateMaterial.C src/MaterialParCartesianVels.C
    src/MaterialParCartesianVp.C src/MParGridFile.C src/MaterialParCartesianVsVp.C
    src/MaterialParAllpts.C src/MaterialParCart.C src/solve-dudp.C src/MaterialParCurv.C)
  target_link_libraries(sw4mopt PRIVATE sw4_native)
endif()
message(STATUS "SW4 backend: ${SW4_BACKEND}; precision: ${SW4_PRECISION}")
include(CTest)
if(BUILD_TESTING)
  include(cmake/SW4Tests.cmake)
endif()

include(GNUInstallDirs)
install(TARGETS sw4 RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(SW4_BUILD_MOPT)
  install(TARGETS sw4mopt RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
